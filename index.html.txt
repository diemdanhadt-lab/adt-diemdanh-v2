/* Người của tổ: danh sách gốc + người thêm − người đã xóa. Mã người = tổ|nhóm|tên */
function people(tid){
  const out=[], rm=DB.removed[tid]||[];
  const add=(g,item)=>{
    const [nm,note=''] = item.split('|').map(s=>s.trim());
    if(!nm) return;
    const code = tid+'|'+g.n+'|'+nm;
    if(rm.includes(code)) return;
    out.push({code, name:nm, note, g});
  };
  team(tid).groups.forEach(g=>{
    g.w.split(',').map(s=>s.trim()).filter(Boolean).forEach(i=>add(g,i));
    (DB.extra[tid]||[]).filter(e=>e.g===g.n).forEach(e=>add(g,e.n));
  });
  return out;
}

  return Math.min(40, Math.max(3, last+2));
}

function supAgg(days,tid){

  days.forEach(d=>{
    for(let i=0;i<40;i++){
      const r=getRec(d,tid+'|HT|'+i); if(!r.nm) continue;

    }
  });

}
/* Tỷ lệ làm: tính trên buổi đã chấm (Làm / (Làm+Phép+K.phép)); ngày chưa chấm không tính là nghỉ */
function agg(tid,days){
  let L=0,P=0,K=0;
  const ids = tid ? [tid] : TEAMS.map(t=>t.id);
  ids.forEach(id=>people(id).forEach(p=>days.forEach(d=>{
    const r=getRec(d,p.code);
    L+=(r.sL?1:0)+(r.cL?1:0); P+=(r.sP?1:0)+(r.cP?1:0); K+=(r.sK?1:0)+(r.cK?1:0);
  })));
  const tot=L+P+K;
  return {L,P,K,tot,rate: tot ? L/tot*100 : null};
}

/* ===== 9. BIỂU ĐỒ & XẾP HẠNG (quản trị) ===== */
function spark(tid,days){
  const pts=[];
  days.forEach((d,i)=>{ const r=agg(tid,[d]).rate; if(r!=null) pts.push([i*(78/Math.max(1,days.length-1))+1, 19-(r/100)*18]); });
  if(pts.length<2) return '–';
  return `<svg width="80" height="20"><polyline fill="none" stroke="var(--pri)" stroke-width="2" points="${pts.map(p=>p.join(',')).join(' ')}"/></svg>`;
}
function barSvg(days){
  const rows=days.map(d=>{
    let L=0,P=0,K=0,S2=0;
    TEAMS.forEach(t=>people(t.id).forEach(p=>{
      const r=getRec(d,p.code);
      L+=(r.sL?1:0)+(r.cL?1:0); P+=(r.sP?1:0)+(r.cP?1:0); K+=(r.sK?1:0)+(r.cK?1:0); S2+=2;
    }));
    return {L,P,K,U:S2-L-P-K};
  });
  const W=680, H=180, max=Math.max(1,...rows.map(r=>r.L+r.P+r.K+r.U));
  const bw=(W-30)/days.length;
  let s=`<svg viewBox="0 0 ${W} ${H+20}" width="100%">`;
  rows.forEach((r,i)=>{
    let y=H;
    [['L','var(--pri)'],['P','#b54708'],['K','var(--red)'],['U','#9aa6a0']].forEach(([k,c])=>{
      const h=r[k]/max*H; y-=h;
      s+=`<rect x="${15+i*bw}" y="${y}" width="${bw*0.7}" height="${h}" fill="${c}"/>`;
    });
    if(i%Math.ceil(days.length/8)===0) s+=`<text x="${15+i*bw}" y="${H+14}" font-size="10" fill="var(--mut)">${days[i].slice(8)}</text>`;
  });
  return s+'</svg>';
}
function heat(days){
  const c = r => r==null ? '' : r>=95 ? 'hg' : r>=90 ? 'hy' : 'hr';
  const row=(name,tid,bold)=>`<tr><td class="nm">${bold?'<b>'+name+'</b>':name}</td>${days.map(d=>{
    const r=agg(tid,[d]).rate; return `<td class="${c(r)}">${r==null?'–':fmtNum(Math.round(r))}</td>`; }).join('')}</tr>`;
  return `<div class="wrap"><table><thead><tr><th class="nm">Tổ</th>${days.map(d=>`<th>${d.slice(8)}</th>`).join('')}</tr></thead>
    <tbody>${TEAMS.map(t=>row(t.name,t.id,false)).join('')}${row('Cả công ty',null,true)}</tbody></table></div>`;
}
function renderChart(){
  const {cur,prev}=periods();
  const co=agg(null,cur), cp=agg(null,prev);
  const rk=TEAMS.map(t=>({t, c:agg(t.id,cur), p:agg(t.id,prev)}));
  const byC=[...rk].sort((a,b)=>(b.c.rate??-1)-(a.c.rate??-1));
  const byP=[...rk].sort((a,b)=>(b.p.rate??-1)-(a.p.rate??-1));
  const ch=rk.map(x=>({x, d:dlt(x.c.rate,x.p.rate)})).filter(o=>o.d!=null);
  const up=[...ch].sort((a,b)=>b.d-a.d)[0], dn=[...ch].sort((a,b)=>a.d-b.d)[0];
  const kr=rk.filter(x=>x.c.tot).sort((a,b)=>b.c.K/b.c.tot-a.c.K/a.c.tot)[0];
  const share=(v,t)=> t ? fmtNum((v/t*100).toFixed(1))+'%' : '–';
  const nTotal=TEAMS.reduce((s,t)=>s+people(t.id).length,0);
  const pbtn=[[7,'7 ngày'],[14,'14 ngày'],[30,'30 ngày'],['month','Tháng']]
    .map(([v,l])=>`<button class="btn sm ${period===v?'pri':''}" onclick="setPeriod('${v}')">${l}</button>`).join('');
  const rankRows=byC.map((x,i)=>{
    const mv=byP.findIndex(y=>y.t.id===x.t.id)-i;
    return `<tr><td>${i+1} ${mv>0?'▲'+mv:mv<0?'▼'+(-mv):'–'}</td><td class="nm">${x.t.name}</td>
      <td>${pct(x.c.rate)}</td><td>${share(x.c.L,x.c.tot)}</td><td>${share(x.c.P,x.c.tot)}</td><td>${share(x.c.K,x.c.tot)}</td>
      <td>${pct(x.p.rate)}</td><td>${arrow(dlt(x.c.rate,x.p.rate))}</td><td>${spark(x.t.id,cur)}</td><td>${lab(x.c.rate)}</td></tr>`;
  }).join('');
  const coRow=`<tr class="tot"><td>–</td><td class="nm">Cả công ty</td><td>${pct(co.rate)}</td>
    <td>${share(co.L,co.tot)}</td><td>${share(co.P,co.tot)}</td><td>${share(co.K,co.tot)}</td>
    <td>${pct(cp.rate)}</td><td>${arrow(dlt(co.rate,cp.rate))}</td><td></td><td>${lab(co.rate)}</td></tr>`;
  document.getElementById('chartBody').innerHTML = `
    <div class="row">${pbtn}</div>
    <div class="stat"><b>Tổng nhân sự</b>${nTotal} người</div>
    <div class="stat"><b>Tỷ lệ làm</b>${pct(co.rate)} · ${arrow(dlt(co.rate,cp.rate))} so với kỳ trước</div>
    <div class="stat"><b>Phép</b>${share(co.P,co.tot)} · <b>Không phép</b>${share(co.K,co.tot)}</div>
    <div class="stat"><b>⬆ Tăng mạnh nhất:</b>${up?up.x.t.name+' '+arrow(up.d):'–'}
      · <b>⬇ Giảm mạnh nhất:</b>${dn?dn.x.t.name+' '+arrow(dn.d):'–'}
      · <b>Nghỉ không phép cao nhất:</b>${kr?kr.t.name+' ('+share(kr.c.K,kr.c.tot)+')':'–'}</div>
    <p class="hint">Xếp hạng theo tỷ lệ làm, kỳ đến ngày ${fmtDate(curDay)}.</p>
    <div class="wrap"><table>
      <thead><tr><th>Hạng</th><th class="nm">Tổ</th><th>Tỷ lệ làm</th><th>Làm</th><th>Phép</th><th>K.phép</th><th>Kỳ trước</th><th>Biến động</th><th>Xu hướng</th><th>Đánh giá</th></tr></thead>
      <tbody>${rankRows}${coRow}</tbody></table></div>
    <p class="hint">Nhân sự theo ngày toàn công ty:</p>
    <div class="wrap" style="padding:8px">${barSvg(cur)}
      <div class="hint">■ <span style="color:var(--pri)">Làm</span> ■ <span style="color:#b54708">Phép</span> ■ <span style="color:var(--red)">K.phép</span> ■ <span style="color:#9aa6a0">Chưa chấm</span></div></div>
    <p class="hint">Tỷ lệ làm (%) theo ngày. Xanh từ 95, vàng 90–95, đỏ dưới 90:</p>
    ${heat(cur)}`;

function importJson(f){
  if(!f) return;
  const rd=new FileReader();
  rd.onload=()=>{
    if(!confirm('Nhập sẽ THAY TOÀN BỘ dữ liệu hiện có. Tiếp tục?')) return;
    try{ DB=Object.assign({}, DEF, JSON.parse(rd.result)); log('Nhập dữ liệu từ JSON'); save(); }
    catch(e){ alert('File không hợp lệ.'); }
  };
  rd.readAsText(f);
}

function addSheet(wb,tid,mode){
  const days = mode==='day' ? [curDay] : monthDays(curDay);
  const ws = wb.addWorksheet(team(tid).name.slice(0,31));
  ws.pageSetup = {orientation:'landscape', paperSize:9, fitToPage:true, fitToWidth:1, fitToHeight:0};
  ws.mergeCells('A1:N1');
  ws.getCell('A1').value = 'ĐIỂM DANH TỔ SẢN XUẤT – ADT · '+team(tid).name+' · '+
    (mode==='day' ? 'Ngày '+fmtDate(curDay) : 'Tháng '+curDay.slice(5,7)+'/'+curDay.slice(0,4));
  ws.getCell('A1').font = {bold:true, size:13};
  const hr = ws.addRow(['STT','Họ và tên','Sáng Làm','Sáng Phép','Sáng K.phép','Chiều Làm','Chiều Phép','Chiều K.phép','Đi trễ','Ra ngoài','Xuống ca','Tăng ca','Giờ thực','Ghi chú']);
  hr.font = {bold:true};
  hr.alignment = {horizontal:'center', vertical:'middle', wrapText:true};
  ws.columns = [6,28,10,10,11,10,10,11,9,10,10,10,14,24].map(w=>({width:w}));
  const first = hr.number+1;
  const rows = [];
  const gtOf = c => days.length===1 ? (getRec(days[0],c).gt||'') : (days.filter(d=>getRec(d,c).gt).length || '');
  const gcOf = c => days.length===1 ? (getRec(days[0],c).gc||'') : '';
  people(tid).forEach(p=>rows.push({name:p.name+(p.note?' ('+p.note+')':''), c:cntOf(days,p.code), gt:gtOf(p.code), gc:gcOf(p.code)}));
  if(mode==='day'){
    for(let i=0;i<supRows(curDay,tid);i++){
      const c=tid+'|HT|'+i, r=getRec(curDay,c);
      if(r.nm) rows.push({name:r.nm+' (hỗ trợ)', c:cntOf([curDay],c), gt:r.gt||'', gc:r.gc||''});
    }
  } else {
    supAgg(days,tid).forEach(s=>rows.push({name:s.name+' (hỗ trợ)', c:s.c, gt:'', gc:''}));
  }
  for(let k=0;k<3;k++) rows.push({name:'', c:{}, gt:'', gc:''});   // dòng trống cho tổ viết thêm
  rows.forEach((r,i)=>{
    const vals = FIELDS.map(f => mode==='day' ? (r.c[f]?'✓':'') : (r.c[f]||0));
    const row = ws.addRow([r.name?i+1:'', r.name, ...vals, r.gt, r.gc]);
    row.alignment = {horizontal:'center', vertical:'middle'};
    row.getCell(2).alignment = {horizontal:'left', vertical:'middle'};
  });
  const last = first+rows.length-1;
  const tr = ws.addRow(['', 'TỔNG CỘNG']);
  tr.font = {bold:true};
  for(let c=3;c<=12;c++){
    const col=String.fromCharCode(64+c), rg=col+first+':'+col+last;
    tr.getCell(c).value = {formula: mode==='day' ? 'COUNTIF('+rg+',"✓")' : 'SUM('+rg+')'};
    tr.getCell(c).alignment = {horizontal:'center'};
  }
}
async function exportXlsx(mode, all){
  if(typeof ExcelJS==='undefined'){ alert('Chưa tải được thư viện Excel. Kiểm tra mạng rồi thử lại.'); return; }
  const wb = new ExcelJS.Workbook();
  (all ? TEAMS : [team(viewTeam)]).forEach(t=>addSheet(wb,t.id,mode));
  const buf = await wb.xlsx.writeBuffer();
  const url = URL.createObjectURL(new Blob([buf],{type:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'}));
  const a = document.createElement('a');
  a.href = url;
  a.download = (all?'ADT_CongTy_':'ADT_'+viewTeam+'_')+(mode==='day'?curDay:curDay.slice(0,7))+'.xlsx';
  a.click();
  setTimeout(()=>URL.revokeObjectURL(url),1000);
}
