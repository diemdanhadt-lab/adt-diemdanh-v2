const LOCK = {sL:['sP','sK'], sP:['sL'], sK:['sL'], cL:['cP','cK'], cP:['cL'], cK:['cL']};
const CHECK_FIELDS = ['sL','sP','sK','cL','cP','cK'];
const TICK_FIELDS = ['tre','ra','xc','tc'];

/* ===== 2. LƯU TRỮ CỤC BỘ (Firebase sẽ thêm ở Bước 3) ===== */
let DB = JSON.parse(localStorage.getItem('adt_v1') || '{"pins":{},"att":{},"done":{}}');
function save(){ localStorage.setItem('adt_v1', JSON.stringify(DB)); }

/* ===== 3. TIỆN ÍCH ===== */
let session = null;      // tổ đang đăng nhập
let curDay = todayKey();

function todayKey(){
  const d = new Date();
  return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
}
function fmtDate(k){ const [y,m,d]=k.split('-'); return d+'/'+m+'/'+y; }   // dd/mm/yyyy
async function sha(s){
  const b = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(s));
  return [...new Uint8Array(b)].map(x=>x.toString(16).padStart(2,'0')).join('');
}
function team(){ return TEAMS.find(t=>t.id===session); }

/* Danh sách người của tổ, mã người = tổ|nhóm|tên */
function people(){
  const out = [];
  team().groups.forEach(g=>{
    g.w.split(',').map(s=>s.trim()).filter(Boolean).forEach(item=>{
      const [nm, note=''] = item.split('|').map(s=>s.trim());
      out.push({code: team().id+'|'+g.n+'|'+nm, name:nm, note, group:g});
    });
  });
  return out;
}
function dayRec(){
  DB.att[curDay] = DB.att[curDay] || {};
  return DB.att[curDay];
}

/* ===== 4. ĐĂNG NHẬP ===== */
function showLogin(){
  document.getElementById('login').classList.remove('hidden');
  document.getElementById('app').classList.add('hidden');
  const sel = document.getElementById('loginTeam');
  sel.innerHTML = TEAMS.map(t=>`<option value="${t.id}">${t.name}${DB.pins[t.id]?'':' (chưa có PIN)'}</option>`).join('');
}
async function doLogin(){
  const id = document.getElementById('loginTeam').value;
  const pin = document.getElementById('loginPin').value.trim();
  const msg = document.getElementById('loginMsg');
  if(!pin){ msg.textContent='Nhập mã PIN.'; return; }
  const h = await sha(pin);
  if(!DB.pins[id]){            // tổ chưa có PIN: đặt PIN mới
    DB.pins[id] = h; save();
  } else if(DB.pins[id] !== h){
    msg.textContent='Mã PIN không đúng.'; return;
  }
  session = id;
  localStorage.setItem('adt_session', id);
  document.getElementById('loginPin').value='';
  msg.textContent='';
  openApp();
}
function logout(){ session=null; localStorage.removeItem('adt_session'); showLogin(); }

/* ===== 5. NGÀY ===== */
function setDay(k){ if(!k) return; curDay=k; render(); }
function shiftDay(n){
  const [y,m,d]=curDay.split('-').map(Number);
  const dt = new Date(y,m-1,d+n);
  setDay(dt.getFullYear()+'-'+String(dt.getMonth()+1).padStart(2,'0')+'-'+String(dt.getDate()).padStart(2,'0'));
}

/* ===== 6. CHẤM CÔNG ===== */
function toggle(code, f){
  const r = dayRec(); r[code] = r[code] || {};
  const p = r[code];
  if(p[f]){ p[f]=false; }
  else{
    if((LOCK[f]||[]).some(g=>p[g])) return;   // bị khóa
    p[f]=true;
  }
  save(); render();
}
function setText(code, f, v){
  const r = dayRec(); r[code] = r[code] || {};
  r[code][f] = v.trim(); save();
}
function allWork(){                           // chưa chấm → làm cả ngày
  const r = dayRec();
  people().forEach(p=>{
    const s = r[p.code] = r[p.code] || {};
    if(!CHECK_FIELDS.some(f=>s[f])){ s.sL=true; s.cL=true; }
  });
  save(); render();
}
function markDone(){
  DB.done[curDay+'|'+session] = true; save(); render();
}

/* ===== 7. HIỂN THỊ ===== */
function openApp(){
  document.getElementById('login').classList.add('hidden');
  document.getElementById('app').classList.remove('hidden');
  document.getElementById('who').textContent = '👤 Tổ: ' + team().name;
  render();
}

function render(){
  document.getElementById('dayPick').value = curDay;
  document.getElementById('dayLabel').textContent = fmtDate(curDay);
  const r = dayRec();
  const tb = document.getElementById('tbody');
  let html = '', lastGroup = null, stt = 0;
  const cnt = {total:0, L:0, P:0, K:0, none:0};

  people().forEach(p=>{
    if(p.group.n !== lastGroup){
      lastGroup = p.group.n;
      html += `<tr class="grp"><td colspan="14">${p.group.n} – ${p.group.t==='TV'?'Thời vụ':'Chính thức'}</td></tr>`;
    }
    const s = r[p.code] || {};
    stt++; cnt.total++;
    const anyL = s.sL||s.cL, anyP = s.sP||s.cP, anyK = s.sK||s.cK;
    if(anyL) cnt.L++; if(anyP) cnt.P++; if(anyK) cnt.K++;
    if(!CHECK_FIELDS.some(f=>s[f])) cnt.none++;

    const cell = f=>{
      const dis = (LOCK[f]||[]).some(g=>s[g]) && !s[f];
      return `<td><div class="cell ${s[f]?'on':''} ${dis?'dis':''}" onclick="toggle('${p.code}','${f}')">${s[f]?'✓':''}</div></td>`;
    };
    const tick = f=>`<td><div class="cell ${s[f]?'on':''}" onclick="toggle('${p.code}','${f}')">${s[f]?'✓':''}</div></td>`;

    html += `<tr>
      <td class="stt">${stt}</td>
      <td class="nm">${p.name}${p.note?` <span class="note">(${p.note})</span>`:''}</td>
      ${cell('sL')}${cell('sP')}${cell('sK')}
      ${cell('cL')}${cell('cP')}${cell('cK')}
      ${TICK_FIELDS.map(tick).join('')}
      <td><input class="txt" value="${s.gt||''}" onchange="setText('${p.code}','gt',this.value)" placeholder="17:00-19:00"></td>
      <td><input class="txt" value="${s.gc||''}" onchange="setText('${p.code}','gc',this.value)"></td>
    </tr>`;
  });
  tb.innerHTML = html;

  document.getElementById('summary').innerHTML =
    `<b>Tổng: ${cnt.total} người</b>
     <span class="c-ok"><b>✓ Làm</b>${cnt.L}</span>
     <span class="c-p"><b>Phép</b>${cnt.P}</span>
     <span class="c-kp"><b>K.phép</b>${cnt.K}</span>
     <span><b>Chưa chấm</b>${cnt.none}</span>`;

  const done = DB.done[curDay+'|'+session];
  document.getElementById('doneMsg').textContent = done
    ? '✅ Tổ đã bấm Lưu – chấm xong cho ngày '+fmtDate(curDay)
    : 'Chưa bấm Lưu – chấm xong';
}
